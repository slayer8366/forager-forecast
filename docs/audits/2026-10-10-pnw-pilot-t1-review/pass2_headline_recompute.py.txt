"""Reviewer pass 2, items 2 to 4: unit sets, the headline skill and interval, the per-fold table.

Own code: reads the held-out predictions in each <prefix>_{calendar,full}.npz and recomputes
Brier, pooled skill (1 - Brier_full / Brier_calendar), a 95% interval from 1,000 resamples of
cells (own implementation, numpy PCG64 seeded 1, not the builder's seed or function), AUC by rank,
and the per-fold table. Compares each figure with the builder's <prefix>_comparison.json.
Run: uv run python headline_recompute.py <fits dir> <prefix> [<prefix> ...]
"""
import json, sys
from pathlib import Path
import numpy as np
from scipy.stats import rankdata

def brier(p, y): return float(np.mean((p - y) ** 2))
def auc(p, y):
    r = rankdata(p); n1 = y.sum(); n0 = len(y) - n1
    return float((r[y == 1].sum() - n1 * (n1 + 1) / 2) / (n1 * n0))

F = Path(sys.argv[1])
for prefix in sys.argv[2:]:
    c = np.load(F / f"{prefix}_calendar.npz"); f = np.load(F / f"{prefix}_full.npz")
    same = bool(np.array_equal(c["unit"], f["unit"]) and np.array_equal(c["y"], f["y"]))
    y, pc, pf, yr, cell = c["y"], c["p"], f["p"], c["year"], c["cell"]
    yr_from_unit = np.array([int(u.split("@")[1][:4]) for u in c["unit"]])
    cell_from_unit = np.array([u.split("@")[0] for u in c["unit"]])
    bc, bf = brier(pc, y), brier(pf, y)
    skill = 1 - bf / bc
    # cluster bootstrap by cell: resample cells with replacement, sum squared errors per cell
    cells, inv = np.unique(cell, return_inverse=True)
    sc = np.bincount(inv, (pc - y) ** 2); sf = np.bincount(inv, (pf - y) ** 2)
    rng = np.random.Generator(np.random.PCG64(1))
    k = len(cells); draws = []
    for _ in range(1000):
        idx = rng.integers(0, k, k)
        draws.append(1 - sf[idx].sum() / sc[idx].sum())
    lo, hi = np.percentile(draws, [2.5, 97.5])
    b = json.loads((F / f"{prefix}_comparison.json").read_text())
    print(f"== {prefix}")
    print(f"  same units and labels in both runs: {same}; unit year == fold year: "
          f"{bool((yr_from_unit == yr).all())}; unit cell == cell: {bool((cell_from_unit == cell).all())}")
    print(f"  units {len(y)} positives {int(y.sum())} cells {k} (builder {b['units']} {b['positives']} {b['cells']})")
    if "land_point" in f.files:
        lp = f["land_point"]
        print(f"  land_point classes: { {s: int((lp == s).sum()) for s in np.unique(lp)} }; "
              f"positives on neighbour units {int(y[lp == 'neighbour'].sum())}")
    print(f"  brier calendar {bc:.9f} full {bf:.9f} (builder {b['brier_calendar']:.9f} {b['brier_full']:.9f})")
    print(f"  skill {skill:.6f} (builder {b['skill']:.6f})")
    print(f"  my 95% cell-resample interval [{lo:.6f}, {hi:.6f}] (builder {b['interval_95'][0]:.6f}, "
          f"{b['interval_95'][1]:.6f}); share of my draws below 0: {np.mean(np.array(draws) < 0):.3f}")
    print(f"  auc calendar {auc(pc, y):.6f} full {auc(pf, y):.6f} (builder {b['auc_calendar']:.6f} {b['auc_full']:.6f})")
    print(f"  per fold (year units positives informative>=30 brier_cal brier_full skill | builder match):")
    bf_by = {d["year"]: d for d in b["folds"]}
    for Y in sorted(set(yr.tolist())):
        m = yr == Y; pos = int(y[m].sum())
        s = 1 - brier(pf[m], y[m]) / brier(pc[m], y[m])
        d = bf_by[Y]
        ok = (d["units"] == int(m.sum()) and d["positives"] == pos and d["informative"] == (pos >= 30)
              and abs(d["skill"] - s) < 1e-12)
        print(f"    {Y} {int(m.sum()):6d} {pos:4d} {str(pos >= 30):5s} {brier(pc[m], y[m]):.6f} "
              f"{brier(pf[m], y[m]):.6f} {s:+.6f} | {ok}")
    inf = np.isin(yr, [Y for Y in set(yr.tolist()) if y[yr == Y].sum() >= 30])
    print(f"  pooled skill over informative folds only: {1 - brier(pf[inf], y[inf]) / brier(pc[inf], y[inf]):+.6f} "
          f"({int(inf.sum())} units)")
