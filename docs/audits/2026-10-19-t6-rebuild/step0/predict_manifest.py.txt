"""Step 0: rebuild each track's effort_manifest.json from ef3269f's code (scripts/t6_fit.py:233-251)
and the committed results' final configuration and frame size, without reading any records,
and hash it against stored_surface_sha256.txt. Run with cwd = the ef3269f worktree."""
import hashlib, json, subprocess, sys
from pathlib import Path
from forager_forecast import effort as ef

audit = Path(sys.argv[1])  # committed docs/audits/2026-10-06-t6-fit on main
filed = dict(reversed(l.split()) for l in (audit / "stored_surface_sha256.txt").read_text().splitlines())
commit = subprocess.run(["git", "rev-parse", "HEAD"], capture_output=True, text=True).stdout.strip()
doi = json.loads(Path("docs/pulls/gbif-fungi-us-canada-2015-2025.doi.json").read_text())
grid = {c.id: c for c in ef.config_grid()} if hasattr(ef, "config_grid") else None
for track in ("all", "cc"):
    r = json.loads((audit / f"{track}_results.json").read_text())
    config = next(ef.Config(p, s) for p in ef.GRID_PULL for s in ef.GRID_SMOOTHING
                  if ef.Config(p, s).id == r["final_configuration"])
    manifest = {
        "task": "T6 effort surface (D101 to D108)",
        "licence_track": {"all": "all licences (D61)", "cc": "CC0_1_0 and CC_BY_4_0 only (D108)"}[track],
        "input_doi": doi["doi"],
        "input_zip_sha256": doi["sha256_of_zip"],
        "code_commit": commit,
        "configuration": {"pull_pseudo_weeks": config.pull, "season_smoothing_weeks": config.smoothing},
        "seed": ef.SEED,
        "frame_cells": r["frame_cells"],
        "frame_weeks": f"{ef.frame_weeks()[0].id} to {ef.frame_weeks()[-1].id}",
        "reads": "effort = cell_level.all_fungi x year_level.effort x season.effort (band, slot) x "
        "sum over day_type of all_fungi_per_day x days; forager_forecast.effort.EffortSurface",
    }
    text = json.dumps(manifest, indent=2) + "\n"
    got = hashlib.sha256(text.encode()).hexdigest()
    want = filed[f"data/t6/{track}/effort_manifest.json"]
    print(track, got, "MATCH" if got == want else f"DIFFERS from filed {want}")
    print(text)
