"""For each named request record: find the store's job for that exact request, read the size the
store states for its result, and compare with the file on disk; then read the file's stamps.
Run: uv run --with cdsapi==0.7.7 python -I publisher_check.py <scripts dir> <file.nc> <request.json> ..."""
import hashlib, json, sys
from pathlib import Path
sys.path.insert(0, sys.argv[1])
import cds_jobs, cdsapi, h5py, numpy as np
c = cdsapi.Client(quiet=True, progress=False)
h = {"PRIVATE-TOKEN": c.key}
args = sys.argv[2:]
out = []
for nc, rq in zip(args[::2], args[1::2]):
    rec = json.loads(Path(rq).read_text())
    job = rec.get("adopted_job") or cds_jobs.existing_job(c, rec["dataset"], rec["request"])
    r = c.session.get(f"{c.url}/retrieve/v1/jobs/{job}/results", headers=h, timeout=60); r.raise_for_status()
    asset = r.json()["asset"]["value"]
    size = Path(nc).stat().st_size
    with h5py.File(nc) as f:
        t = np.atleast_1d(f["valid_time"][()]).astype(np.int64)
        ev = sorted({(v.decode() if isinstance(v, bytes) else str(v)) for v in np.atleast_1d(f["expver"][()])}) if "expver" in f else None
        shape = {k: list(f[k].shape) for k in f if k not in ("valid_time","expver","latitude","longitude","number")}
        grid = [float(f["latitude"][0]), float(f["latitude"][-1]), float(f["longitude"][0]), float(f["longitude"][-1])]
    out.append({"file": Path(nc).name, "job": job, "store_file_size": asset.get("file:size"),
        "store_type": asset.get("type"), "disk_bytes": size, "size_match": asset.get("file:size") == size,
        "sha256": hashlib.sha256(Path(nc).read_bytes()).hexdigest(),
        "stamps": len(t), "first_stamp": str(np.datetime64(int(t.min()), "s")), "last_stamp": str(np.datetime64(int(t.max()), "s")),
        "expver": ev, "shapes": shape, "grid_lat_lon_edges": grid})
print(json.dumps(out, indent=1))
sys.exit(0 if all(o["size_match"] for o in out) else 1)
