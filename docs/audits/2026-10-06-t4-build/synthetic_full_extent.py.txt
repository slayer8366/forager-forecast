"""Synthetic full-extent run (no ISRIC data): noise sources over the whole D82 rectangle.

Measures (1) the regrid's run time for the real test area, (2) the sampling error of the ten-cell
check's point sampler against the exact regrid on worst-case data (independent noise per native
pixel), which sizes TOLERANCE_PH, and (3) runs the ten-cell script on the synthetic outputs as a
positive control. Run: uv run python THIS DATA_DIR (data dir under /tmp, not the repository).
"""
import json, math, sys, time
from pathlib import Path
import numpy as np
import rasterio
from rasterio.transform import from_origin

sys.path.insert(0, str(Path(__file__).resolve().parents[3] / "scripts")) if False else None
from forager_forecast.soilgrids import HOMOLOSINE, layer_name, layers, native_bounds_for
from forager_forecast.t4_layer import WESTERN_WASHINGTON, build_archive, build_master, fetch_natives, master_window

out = Path(sys.argv[1]); out.mkdir(parents=True, exist_ok=True)
window = master_window(WESTERN_WASHINGTON)
left, bottom, right, top = native_bounds_for(window, margin_m=3000.0)
x0 = math.floor(left / 250) * 250 - 37.0; y0 = math.ceil(top / 250) * 250 + 61.0
w = math.ceil((right - x0) / 250) + 2; h = math.ceil((y0 - bottom) / 250) + 2
rng = np.random.default_rng(20260918)
src_dir = out / "isric"; src_dir.mkdir(exist_ok=True)
sources = {}
for depth, stat in layers():
    base = {"mean": 55, "Q0.05": 45, "Q0.95": 65}[stat]
    v = (base + rng.integers(-15, 16, (h, w))).astype("int16")
    v[: h // 5, : w // 6] = -32768  # a no-data block (stands in for ocean)
    p = src_dir / f"{layer_name(depth, stat)}.tif"
    with rasterio.open(p, "w", driver="GTiff", width=w, height=h, count=1, dtype="int16", crs=HOMOLOSINE,
                       transform=from_origin(x0, y0, 250.0, 250.0), nodata=-32768, compress="deflate") as d:
        d.write(v, 1)
    sources[(depth, stat)] = str(p)
t0 = time.perf_counter()
fetch_natives(out / "native", WESTERN_WASHINGTON, source_for=lambda d, s: sources[(d, s)])
t1 = time.perf_counter()
summary = build_master(out / "native", out / "master.tif", WESTERN_WASHINGTON)
t2 = time.perf_counter()
header, meta = build_archive(out / "master.tif", out / "archive")
t3 = time.perf_counter()
print(json.dumps({"window": summary["window"], "cells": summary["cells"], "regrid_seconds": round(summary["regrid_seconds"], 1),
                  "build_master_seconds": round(t2 - t1, 1), "archive_seconds": round(t3 - t2, 1),
                  "header_zoom": [header["min_zoom"], header["max_zoom"]]}, indent=1))

# Sampling error of the ten-cell sampler against the exact regrid, on 2,000 random full cells.
sys.path.insert(0, str(Path.cwd() / "scripts"))
import t4_ten_cells as tc
native, nt = tc.load_native(out / "native")
with rasterio.open(out / "master.tif") as m:
    mean = m.read(1); frac = m.read(4); T = m.transform
full = np.argwhere(np.isfinite(mean) & (frac == 1.0))
pick = full[rng.choice(len(full), 2000, replace=False)]
diffs = []
for r, c in pick:
    col = int(round(T.c / 250)) + c; row = int(round(T.f / 250)) - 1 - r
    sv = tc.sampled_value(native, nt, col, row)["mean"][0]
    diffs.append(abs(sv - float(mean[r, c])))
d = np.array(diffs)
print(json.dumps({"sampler_vs_exact_on_noise_n": len(d), "max": round(float(d.max()), 4),
                  "p99": round(float(np.percentile(d, 99)), 4), "mean": round(float(d.mean()), 4)}))
