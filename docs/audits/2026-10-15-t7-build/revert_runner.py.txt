COMPILE_MARKERS = ("SyntaxError", "ImportError", "ModuleNotFoundError", "IndentationError",
                   "errors during collection", "ERROR collecting", "NameError")


def sha(path):
    return hashlib.sha256(Path(path).read_bytes()).hexdigest()


def clear_pycache():
    for d in Path("src").rglob("__pycache__"):
        shutil.rmtree(d)
    for d in Path("tests").rglob("__pycache__"):
        shutil.rmtree(d)


def run_one(name, path, old, new, predicted, scratch):
    text = Path(path).read_text(encoding="utf-8")
    if text.count(old) != 1:
        return {"check": name, "refused": f"old text occurs {text.count(old)} times"}
    saved = scratch / (Path(path).name + ".saved")
    shutil.copyfile(path, saved)
    saved_sha = sha(saved)
    try:
        Path(path).write_text(text.replace(old, new), encoding="utf-8")
        clear_pycache()
        env = {**os.environ, "PYTHONDONTWRITEBYTECODE": "1"}
        module = path.removeprefix("src/").removesuffix(".py").replace("/", ".")
        seen = subprocess.run(
            [sys.executable, "-c", f"import inspect, {module} as m; "
             f"print({new!r} in inspect.getsource(m))"],
            capture_output=True, text=True, env=env,
        )  # fmt: skip
        if seen.stdout.strip() != "True":
            return {"check": name, "refused": "interpreter does not see the edit",
                    "stderr": seen.stderr[-2000:]}  # fmt: skip
        xml = scratch / f"{name.split()[0]}.xml"
        xml.unlink(missing_ok=True)
        proc = subprocess.run(
            [sys.executable, "-m", "pytest", "-q", "-p", "no:cacheprovider",
             f"--basetemp=.pytest-tmp/revert-{name.split()[0]}", f"--junitxml={xml}",
             *sorted(str(p) for p in Path(T).glob("test_t7_*.py"))],
            capture_output=True, text=True, env=env,
        )  # fmt: skip
        log = proc.stdout + proc.stderr
        bad = [m for m in COMPILE_MARKERS if m in log]
        if bad or not xml.exists():
            return {"check": name, "refused": f"build or collection error {bad}",
                    "log_tail": log[-3000:]}  # fmt: skip
        root = ET.parse(xml).getroot()
        suite = root if root.tag == "testsuite" else root.find("testsuite")
        if int(suite.get("errors", 0)):
            return {"check": name, "refused": "errors in the XML", "log_tail": log[-3000:]}
        failed = {}
        for tc in root.iter("testcase"):
            f = tc.find("failure")
            if f is not None:
                failed[tc.get("name")] = (f.get("message") or "").splitlines()[0][:300]
        return {"check": name, "file": path, "edit": [old.strip(), new.strip()],
                "tests_run": int(suite.get("tests")), "failed": failed,
                "predicted": predicted, "bites": sorted(failed) == sorted(predicted)}  # fmt: skip
    finally:
        shutil.copyfile(saved, path)
        clear_pycache()
        restored = sha(path) == saved_sha and old in Path(path).read_text(encoding="utf-8")
        if not restored:
            raise SystemExit(f"{name}: restore failed; saved copy at {saved}")


def main():
    out = Path(sys.argv[1])
    scratch = Path(".pytest-tmp/revert-scratch")
    scratch.mkdir(parents=True, exist_ok=True)
    head_before = {p: sha(p) for p in {c[1] for c in CHECKS}}
    results = [run_one(*c, scratch) for c in CHECKS]
    head_after = {p: sha(p) for p in head_before}
    body = {"files_unchanged_after_all_checks": head_before == head_after, "results": results}
    out.write_text(json.dumps(body, indent=2) + "\n")
    for r in results:
        print(r["check"], "| refused:" + r["refused"] if "refused" in r else
              f"| bites={r['bites']} failed={sorted(r['failed'])}")  # fmt: skip
    print("files unchanged after all checks:", body["files_unchanged_after_all_checks"])


if __name__ == "__main__":
    main()
