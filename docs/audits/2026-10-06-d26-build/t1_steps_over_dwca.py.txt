"""t1_steps() over a DWCA download zip, read in place: steps, survivors, and counts by stage and box."""

import io
import json
import sys
import zipfile
from collections import Counter

from forager_forecast.records.filters import Pipeline, t1_steps
from forager_forecast.records.occurrence import OccurrenceLoader, read_occurrence_rows
from forager_forecast.t1_design import box_of

by_stage_box: Counter = Counter()


def tally(stage, record):
    box = box_of(record.latitude, record.longitude)
    by_stage_box[(stage, box.name if box else "outside")] += 1


loader = OccurrenceLoader()
pipeline = Pipeline(t1_steps(), on_pass=tally)
with zipfile.ZipFile(sys.argv[1]).open("occurrence.txt") as raw:
    rows = read_occurrence_rows(io.TextIOWrapper(raw, encoding="utf-8", newline=""))
    kept = pipeline.run(loader.load(rows))
out = {
    "zip": sys.argv[1],
    "rows_read": loader.rows_read,
    "unloadable": dict(loader.unloadable.most_common()),
    "source": pipeline.source_count,
    "steps": [[c.step, c.before, c.dropped, c.after] for c in pipeline.counts],
    "survivors": len(kept),
    "by_stage_box": {f"{s}|{b}": n for (s, b), n in sorted(by_stage_box.items())},
}
assert loader.rows_read == pipeline.source_count + sum(loader.unloadable.values())
json.dump(out, open(sys.argv[2], "w"), indent=1)
print(json.dumps(out, indent=1))
