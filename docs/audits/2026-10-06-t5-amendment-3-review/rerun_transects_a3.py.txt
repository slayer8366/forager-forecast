def old_verdict(border, w_us, w_ca, resamples=10000):
    """The filed rule, verify report section 7: each null over every row with a finite within step."""
    rng = np.random.default_rng(seam.SEED)
    t_us = seam._null_threshold(np.abs(w_us), rng, resamples)
    t_ca = seam._null_threshold(np.abs(w_ca), rng, resamples)
    b = border[np.isfinite(border)]
    big = float(np.median(np.abs(b)))
    rows = lambda w: int(sum(np.isfinite(r).any() for r in w))
    return dict(n=len(b), B=big, threshold=max(t_us, t_ca), artifact=bool(big > max(t_us, t_ca)),
                null_rows_us=rows(w_us), null_rows_ca=rows(w_ca))

def samples(path, name):
    ds, window = seam._sampler(path)
    with ds:
        band = ds.read(list(ds.descriptions).index(name) + 1).astype("float64")
    border, wus, wca = [], [], []
    for lon in seam.transect_longitudes():
        n, s = seam.sample_latitudes(lon)
        def read(lats):
            out = np.full(len(lats), np.nan)
            for i, lat in enumerate(lats):
                try:
                    r, c = window.pixel_of_cell(seam.cell_for_lonlat(lon, float(lat)))
                except IndexError:
                    continue
                out[i] = band[r, c]
            return out
        b, wu, wc = seam.transect_steps(read(n), read(s))
        border.append(b); wus.append(wu); wca.append(wc)
    return np.array(border), np.array(wus), np.array(wca)

def same(a, b):
    if isinstance(a, dict):
        return a.keys() == b.keys() and all(same(a[k], b[k]) for k in a)
    if isinstance(a, list):
        return len(a) == len(b) and all(same(x, y) for x, y in zip(a, b))
    if isinstance(a, float) and isinstance(b, float) and math.isnan(a) and math.isnan(b):
        return True
    return a == b

print("== 2. D90 re-run against the committed files")

            print(f"   {band:18s} n={r['n_transects']:2d} rows us/ca={r['null_rows_us']}/{r['null_rows_ca']} "
                  f"B={r['border_median_abs']:.5f} thr={r['threshold']:.5f} mean={r['border_mean_signed']:+.4f} "
                  f"ci=[{r['border_mean_signed_ci95'][0]:+.4f},{r['border_mean_signed_ci95'][1]:+.4f}] "
                  f"artifact={r['artifact']}")
    # the drive's own copy of the file, if present
    drive = T5 / committed.name
    if drive.exists():
        print("   drive copy == committed:", drive.read_bytes() == committed.read_bytes())
