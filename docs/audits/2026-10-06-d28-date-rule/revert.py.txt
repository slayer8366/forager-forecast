"""Strict revert runner for records/filters.py (D28 Part 2: D97, D98). Run from the worktree root:
uv run python docs/audits/2026-10-06-d28-date-rule/revert.py.txt <results.jsonl>

Per check: save the file's bytes; make one edit (must match exactly once); clear __pycache__;
confirm a fresh interpreter sees the edited source (sha256); run the test file with a fresh JUnit
XML; refuse on any exit other than 1, any error, or a missing XML; restore from the saved bytes,
never from git; confirm byte-identical and that git diff is empty for the file afterwards.
"""
import hashlib, json, os, pathlib, shutil, subprocess, sys, xml.etree.ElementTree as ET

ROOT = pathlib.Path.cwd()
SRC = ROOT / "src/forager_forecast/records/filters.py"
TESTS = ["tests/test_records_filters.py", "tests/test_records_date_quality.py", "tests/test_records_midnight.py", "tests/test_records_duplicate_keys.py"]
XML = ROOT / ".revert_junit.xml"
OLD_RULED = 'a rule.\n    """\n    return False\n'
CHECKS = [
    ("P1 keep-the-1st reverted: date-only on the 1st dropped", OLD_RULED,
     OLD_RULED.replace("return False", "return record.event_date.day == 1 and record.event_time is None")),
    ("P2 midnight check kept: 00:00:00 on the 1st dropped", OLD_RULED,
     OLD_RULED.replace("return False", "return record.event_date.day == 1 and record.event_time == time(0, 0, 0)")),
    ("P3 T1 list keeps the provisional rule", 'FilterStep("date kept as given (D97, D98)", drops_no_dated_record)',
     'FilterStep("date kept as given (D97, D98)", is_default_date)'),
    ("P4 R6 list keeps the provisional rule", 'FilterStep("date_kept_as_given", drops_no_dated_record)',
     'FilterStep("date_kept_as_given", is_default_date)'),
    ("P5 date step found by the provisional predicate", "if step.drops is drops_no_dated_record]",
     "if step.drops is is_default_date]"),
]

def clear_cache():
    for d in ROOT.rglob("__pycache__"):
        if ".venv" not in d.parts:
            shutil.rmtree(d)

out = open(sys.argv[1], "w")
env = dict(os.environ, PYTHONDONTWRITEBYTECODE="1")
saved = SRC.read_bytes()
for name, old, new in CHECKS:
    text = saved.decode()
    assert text.count(old) == 1, (name, text.count(old))
    edited = text.replace(old, new).encode()
    try:
        SRC.write_bytes(edited)
        clear_cache()
        seen = subprocess.run([sys.executable, "-c",
            "import hashlib,forager_forecast.records.filters as m;print(m.__file__);print(hashlib.sha256(open(m.__file__,'rb').read()).hexdigest())"],
            capture_output=True, text=True, env=env).stdout.split()
        assert seen[0] == str(SRC) and seen[1] == hashlib.sha256(edited).hexdigest(), seen
        if XML.exists():
            XML.unlink()
        r = subprocess.run([sys.executable, "-m", "pytest", "-q", "-p", "no:cacheprovider", *TESTS, f"--junitxml={XML}"],
                           capture_output=True, text=True, env=env)
        if not XML.exists():
            raise SystemExit(f"{name}: no XML, refusing\n{r.stdout[-2000:]}")
        suite = ET.parse(XML).getroot().find("testsuite")
        suite = suite if suite is not None else ET.parse(XML).getroot()
        errors, failures = int(suite.get("errors")), int(suite.get("failures"))
        if r.returncode != 1 or errors:
            raise SystemExit(f"{name}: exit {r.returncode}, errors {errors}, refusing\n{r.stdout[-2000:]}")
        failed = []
        for case in suite.iter("testcase"):
            f = case.find("failure")
            if f is not None:
                failed.append({"test": case.get("name"), "message": (f.get("message") or "")[:300]})
        rec = {"check": name, "exit": r.returncode, "errors": errors, "failures": failures, "failed": failed}
    finally:
        SRC.write_bytes(saved)
        clear_cache()
    assert SRC.read_bytes() == saved
    rec["restored_identical"] = True
    print(json.dumps(rec)); out.write(json.dumps(rec) + "\n"); out.flush()
if XML.exists():
    XML.unlink()
d = subprocess.run(["git", "diff", "--stat", "--", str(SRC)], capture_output=True, text=True).stdout
print("git diff on the file after all checks:", repr(d))
