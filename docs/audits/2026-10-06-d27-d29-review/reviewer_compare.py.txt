"""Compare the reviewer's recount (reviewer_recount.json) with the builder's filed outputs.
Usage: python3 reviewer_compare.py.txt <recount json> <repo root>"""
import csv, json, sys
from pathlib import Path
me = json.load(open(sys.argv[1])); root = Path(sys.argv[2])
ev = root / "docs/audits/2026-10-06-d27-d29/tables_0012112"
ok = True
def check(label, mine, theirs):
    global ok
    same = mine == theirs
    ok &= same
    print(("SAME " if same else "DIFF ") + label, mine if same else (mine, theirs))
# day-of-month table
rows = {r["datasetKey"]: r for r in csv.DictReader(open(ev / "day_of_month_by_dataset.csv"))}
for ds, r in rows.items():
    days = {int(k): v for k, v in me["date_only_by_dataset_day"].get(ds, {}).items()}
    check(f"day table {r['title'][:30]} loaded/n/days/midnight",
          (me["loaded_by_dataset"].get(ds, 0), sum(days.values()), [days.get(d, 0) for d in range(1, 32)], me["midnight_first_clock_by_dataset"].get(ds, 0)),
          (int(r["loaded"]), int(r["date_only"]), [int(r[f"day_{d}"]) for d in range(1, 32)], int(r["midnight_first_with_clock"])))
check("datasets in day table", sorted(me["loaded_by_dataset"]), sorted(rows))
n_all = sum(sum(c.values()) for c in me["date_only_by_dataset_day"].values())
k_all = sum(c.get("1", 0) for c in me["date_only_by_dataset_day"].values())
print("whole download date-only", n_all, "on the 1st", k_all, "midnight-on-1st clock", sum(me["midnight_first_clock_by_dataset"].values()))
# survivors
for L, own in (("t1", "event"), ("r6", "observer")):
    for rule, f in (("drop_all", "drop_all"), ("keep_all", "keep_all"), ("keep_all_but_too_few", "per_dataset_plus_too_few")):
        p = json.load(open(ev / f"pass_{L}_{f}.json"))
        for key in ("event", "observer"):
            check(f"{L} {rule} {key} survivors", me[rule][L][key], p["survivors"][key])
    p = json.load(open(ev / f"pass_{L}_drop_all.json"))
    check(f"{L} drop_all entered duplicate step by region", me["entered_duplicate_step_drop_all"][L], p["duplicate_step_reads"]["by_region"])
    check(f"{L} empty recordedBy entered", me["entered_empty_recordedBy_drop_all"].get(L, 0), p["duplicate_step_reads"]["records_with_empty_recordedBy"])
    pk = json.load(open(ev / f"pass_{L}_keep_all.json"))
    check(f"{L} keep_all reach duplicate step = drop_all + extras", sum(me["entered_duplicate_step_drop_all"][L].values()) + me["extra_records_keep_all"][L], pk["duplicate_step_reads"]["records"])
# D29
d = json.load(open(root / "docs/pulls/gbif-fungi-us-canada-2015-2025.datasets.json"))
check("D29 rows by dataset", me["rows_by_dataset"], {x["datasetKey"]: x["records_counted_in_download"] for x in d["datasets"]})
check("D29 rows by licence", me["rows_by_license"], d["records_by_own_license_field"])
cross = {}
for ds, lic, n in me["rows_by_dataset_license"]:
    cross.setdefault(ds, {})[lic] = n
check("D29 dataset x licence", cross, {x["datasetKey"]: x["records_by_own_license_field"] for x in d["datasets"]})
t = sum(me["rows_by_license"].values())
print("licence shares by record", {k: f"{100*v/t:.2f}%" for k, v in me["rows_by_license"].items()}, "rows", me["rows"])
print("unloadable", me["unloadable"], "raw-T vs loader mismatch", me["raw_T_vs_loader_date_only_mismatch"])
print("extras by dataset", me["extra_by_dataset"])
sap = "d1d59f9f-e130-4715-bc9b-8a25e7b62e89"
print("saproxylic date-only on 1st, first failing step:", {L: me["first_failing_step_of_date_only_on_1st"][L].get(sap) for L in ("t1", "r6")})
print("ALL SAME" if ok else "SOME DIFFER")
