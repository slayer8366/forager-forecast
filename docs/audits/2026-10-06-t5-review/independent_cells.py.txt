"""T5 review (D18): genus shares and total cover on real master cells, recomputed independently.

Independent of t5_layer.py, regrid.py and crown_cover.plot_cover: the DBF is parsed here, the
tree table is streamed here, crown cover (Bechtold 2004 Eq. 3, Crookston and Stage eqs 1-2) and
the surrogate rule are re-implemented here, and each master cell is approximated by a 50 x 50
grid of point samples (exact area weighting is not reproduced, so agreement is to about 0.01 in a
share and about 1 point in cover). It reuses only crown_cover.BECHTOLD_2004_EQ3 and _N, which the
review checked row by row against the paper's Tables 3 and 1, and it checks each SPCD it uses
against the tree table's COMMON_NAME. Reads the flash drive only; writes nothing there.
Run: PYTHONDONTWRITEBYTECODE=1 uv run --frozen python <this file>
"""

def dbf(path, want):
    with open(path, "rb") as f:
        n, hlen, rlen = struct.unpack("<IHH", f.read(12)[4:12])
        f.seek(32)
        fields = []
        while (d := f.read(32))[0] != 0x0D:
            fields.append((d[:11].rstrip(b"\0").decode(), d[16]))
        f.seek(hlen)
        out = {w: [] for w in want}
        for _ in range(n):
            r = f.read(rlen)
            p = 1
            for name, size in fields:
                if name in out:
                    out[name].append(r[p:p + size].decode().strip())
                p += size
    return out


def cell_points(col, row, left, top):
    off = (np.arange(SS) + 0.5) / SS * 250
    xs = left + col * 250 + off
    ys = top - row * 250 - off
    return np.meshgrid(xs, ys)

    B = {n: m.read(i + 1) for i, n in enumerate(names)}
    left, top = m.transform.c, m.transform.f

        lon, lat = to_lonlat.transform(gx, gy)

def surrogate(spcd, genus):
    if spcd in EQ3:
        return spcd
    from forager_forecast.crown_cover import TABLE_3_GENUS  # genus labels only
    mates = [s for s, g in TABLE_3_GENUS.items() if g == genus and s not in WOODLAND]
    if mates:
        return max(mates, key=NOBS.get)
    return 202 if genus in GYMNO else 746


def plot_split(rows):
    """(sum p a by genus and class, total p a, zero-width trees, beyond-range trees)."""
    pa = defaultdict(float)
    zero = beyond = 0
    for r in rows:
        if r["STATUSCD"] != "1" or r["DIA"] in ("", "NA") or float(r["DIA"]) < 5.0:
            continue
        tpa = float(r["TPA_UNADJ"]) if r["TPA_UNADJ"] not in ("", "NA") else 0
        if tpa <= 0:
            continue
        spcd, genus, d = int(r["SPCD"]), r["SCIENTIFIC_NAME"].split()[0], float(r["DIA"])
        s = surrogate(spcd, genus)
        b0, b1, b2 = EQ3[s]
        w = b0 + b1 * d + b2 * d * d
        if s in TABLE1_MAX_D and d > TABLE1_MAX_D[s]:
            beyond += 1
        if w <= 0:
            zero += 1
            continue
        a = tpa * math.pi * (w / 2) ** 2
        pa[genus] += a
        pa["conifer" if genus in GYMNO else "broadleaf"] += a
        pa["_total"] += a
    return pa, zero, beyond


split = {p: plot_split(trees.get(p, [])) for p in all_plots}
no_split = sorted(p for p in win_plots if split[p][0]["_total"] == 0 and canopy.get(p, 0) > 0)
no_split_any = sorted(p for p in win_plots if split[p][0]["_total"] == 0)
lat_cut = Transformer.from_crs(crs, "EPSG:4269", always_xy=True).transform(
    *np.meshgrid(t.c + (np.arange(cut.shape[1]) + 0.5) * t.a, t.f + (np.arange(cut.shape[0]) + 0.5) * t.e))[1]

facts = {
    "plots_in_window": len(win_plots),
    "plots_with_canopy_but_no_tree_of_5in": len(no_split),
    "plots_with_no_tree_of_5in_any_canopy": len(no_split_any),
    "forest_pixels_in_cut": int(forest.sum()),
    "forest_pixels_in_cut_south_of_49": int(us_forest.sum()),
    "no_split_pixels_in_cut": int((forest & ns_set).sum()),
    "no_split_pixels_south_of_49": int((us_forest & ns_set).sum()),
    "trees_with_zero_or_negative_width_dropped": sum(split[p][1] for p in win_plots),
    "trees_beyond_table1_max_d_listed_species": sum(split[p][2] for p in win_plots),
    "canopypct_of_no_split_plots": {p: canopy[p] for p in no_split},
}

results = {"facts": facts, "us": [], "canada": []}
for (row, col), (vals, nod) in pts.items():
    n = len(vals)
    tot = sum(canopy[v] if v != nod else 0.0 for v in vals) / n
    num = defaultdict(float)
    den = 0.0
    for v in vals:
        if v == nod:
            continue
        pa = split[v][0]
        if pa["_total"] == 0:
            continue
        den += canopy[v]
        for g in ("Pseudotsuga", "Tsuga", "Abies", "conifer", "broadleaf"):
            num[g] += canopy[v] * pa[g] / pa["_total"]
    mine = {"total_cover_pct": tot, **{f"share_{g}": num[g] / den for g in num}}
    built = {k: float(B[k][row, col]) for k in mine}
    results["us"].append({"row": int(row), "col": int(col), "own_side_samples": n,
                          "independent": mine, "built": built,
                          "max_abs_diff_share": max(abs(mine[k] - built[k]) for k in mine if k.startswith("share")),
                          "abs_diff_cover": abs(mine["total_cover_pct"] - built["total_cover_pct"])})

    with rasterio.open(NATIVE / f"scanfi_{c}.tif") as ds:

for row, col in ca_cells:
    gx, gy = cell_points(col, row, left, top)
    lon, lat = to_lonlat.transform(gx, gy)
    own = lat.ravel() >= 49.0
    x, y = tolcc.transform(gx.ravel()[own], gy.ravel()[own])
    sums = {}
    for c in classes:
        a, tr, _, nod = arr[c]
        cc = np.floor((x - tr.c) / tr.a).astype(int)
        rr = np.floor((y - tr.f) / tr.e).astype(int)
        v = a[rr, cc].astype(float)
        v[v == nod] = 0.0
        sums[c] = v.mean()
    total = sum(sums.values())
    mine = {"total_cover_pct": total, "share_Pseudotsuga": sums["douglasFir"] / total,
            "share_broadleaf": sums["broadleaf"] / total,
            "share_conifer": (total - sums["broadleaf"]) / total}
    built = {k: float(B[k][row, col]) for k in mine}
    results["canada"].append({"row": int(row), "col": int(col), "independent": mine, "built": built,
                              "max_abs_diff_share": max(abs(mine[k] - built[k]) for k in mine if k.startswith("share")),
                              "abs_diff_cover": abs(mine["total_cover_pct"] - built["total_cover_pct"])})

used = sorted({surrogate(int(r["SPCD"]), r["SCIENTIFIC_NAME"].split()[0]) for p in all_plots for r in trees.get(p, [])})
results["facts"]["spcd_common_names_for_table3_species_used"] = {s: names_seen.get(s) for s in used}
