"""T6 review (D18): an independent recount of T6's outings over D26's download, streamed.

Reviewer's own code, rewritten after the first review session's version of this file (7a743d4) was
killed by the kernel's OOM killer at 9.4 GB (Forager RECORD -610). That version held every
surviving record as Python tuples of strings, dates and nested tuples; this one holds only packed
integers: one dict entry per observer-key survivor (key and value each one int), one int set per
outing set, and an int id per distinct observer and dataset string. No cell-week table is built at
any point; the outputs are one row per outing (at most about 0.9 million), never the 49 million
frame rows. RSS is logged every 250,000 rows so growth is visible while it runs.

Imports nothing from forager_forecast. Reads the zip in place, never extracted, splitting
occurrence.txt on tabs (GBIF writes it unquoted). Rules, from D101, D102, D103, D108 and the
loader and step list they cite, re-implemented here:

- unloadable: gbifID or acceptedTaxonKey not an integer; coordinates missing, not finite, or out
  of range; eventDate not one calendar day (empty, month-only, more than one "/", or a range whose
  two ends are different days).
- steps in order: user-obscured (informationWithheld or dataGeneralizations non-empty); coordinate
  uncertainty missing, not finite or above 1,000 m; date step (drops nothing, D97, D98);
  observer-duplicate key (acceptedTaxonKey, recordedBy trimmed, cell, day), lowest gbifID kept.
- cell: nearest 0.1 degree point, exact halves toward +infinity, on the decimal text (D46, D63, D64).
- outing: (observer, cell, day), observer = recordedBy, or "anonymous in dataset X" when empty.
- whole weeks: the 573 ISO weeks 2015-W02 to 2025-W52.
- lichen: classKey 180. CC: licence CC0_1_0 or CC_BY_4_0.

The CC track is counted two ways, because D108's text ("outings for that surface are counted from
those records alone") and the builder's rule ("an outing with at least one CC0 or CC BY survivor of
the all-licence step list", stated to be equal) need not agree: a CC record that loses its
observer key to a lower-gbifID non-CC record is gone from the builder's count but survives a step
list run on CC records alone. Both are written; which one equals 106,765 is read from the output.
The same goes for anonymous records: the observer key holds recordedBy (empty for all of them),
while an outing's observer is "anonymous in dataset X", so two datasets' anonymous records of one
taxon in one cell on one day share a key, and the loser's outing can vanish. Outings before and
after the duplicate step are both counted to measure that.

    python recount.py.txt ZIP OUT_DIR [MAX_ROWS]
"""

import io
import json
import resource
import sys
import time
import zipfile
from collections import Counter
from datetime import date
from decimal import ROUND_FLOOR, Decimal, InvalidOperation
from pathlib import Path

import numpy as np

ZIP, OUT = Path(sys.argv[1]), Path(sys.argv[2])
MAX_ROWS = int(sys.argv[3]) if len(sys.argv) > 3 else None
OUT.mkdir(parents=True, exist_ok=True)
TENTH = Decimal("0.1")
HALF = Decimal("0.5")
CC = {"CC0_1_0", "CC_BY_4_0"}
FIRST_MONDAY = date(2015, 1, 5).toordinal()  # Monday of 2015-W02
LAST_SUNDAY = date(2025, 12, 28).toordinal()  # Sunday of 2025-W52
N_WEEKS = (LAST_SUNDAY - FIRST_MONDAY + 1) // 7
assert N_WEEKS == 573, N_WEEKS
assert date(2015, 1, 5).isocalendar()[:2] == (2015, 2)
assert date(2025, 12, 28).isocalendar()[:2] == (2025, 52)
DAY_BASE = date(2014, 1, 1).toordinal()  # every kept day is after this, packed as an offset


def rss_mb():
    for line in open("/proc/self/status"):
        if line.startswith("VmRSS:"):
            return int(line.split()[1]) // 1024
    return -1


def tenths(text):
    return int((Decimal(text) / TENTH + HALF).to_integral_value(rounding=ROUND_FLOOR))


def one_day(text):
    """The calendar day an eventDate names, "span" for a range over two days, or None."""
    text = text.strip()
    if not text:
        return None
    ends = text.split("/")
    if len(ends) > 2:
        return None
    days = []
    for end in ends:
        head = end.split("T", 1)[0]
        if len(head) != 10:
            return None
        try:
            days.append(date.fromisoformat(head))
        except ValueError:
            return None
    if len(days) == 2 and days[0] != days[1]:
        return "span"
    return days[0]


def finite_float(text):
    text = text.strip()
    if not text:
        return None
    try:
        v = float(text)
    except ValueError:
        return None
    if v != v or v in (float("inf"), float("-inf")):
        return None
    return v


observer_id = {"": 0}  # recordedBy string -> int; 0 is empty
dataset_id = {}
started = time.time()
rows = loaded = 0
unloadable = Counter()
dropped = Counter()
licences = Counter()
class_names_180 = Counter()
anonymous_by_dataset = Counter()  # filtered records with an empty recordedBy
# Observer key -> survivor, packed. Key: taxon, observer id, lat, lon, day offset.
# Value: gbifID << 20 | dataset id << 2 | cc << 1 | lichen.
kept = {}
kept_cc = {}
event_keys = set()  # D27's event key, for its record count only
pre_outings = set()  # outings before the duplicate step, packed


def pack_key(taxon, obs, lat, lon, day):
    assert 0 <= obs < (1 << 24) and -1000 < lat < 1000 and -2000 < lon < 2000
    return ((((taxon << 24) | obs) << 11 | (lat + 1000)) << 12 | (lon + 2000)) << 14 | day


def unpack_key(k):
    day = k & 0x3FFF
    k >>= 14
    lon = (k & 0xFFF) - 2000
    k >>= 12
    lat = (k & 0x7FF) - 1000
    k >>= 11
    obs = k & 0xFFFFFF
    return obs, lat, lon, day


def outing_key(obs, ds, lat, lon, day):
    # observer identity: obs << 16 for a named observer, ds + 1 for an anonymous one
    ident = (obs << 16) if obs else (ds + 1)
    return (((ident << 11) | (lat + 1000)) << 12 | (lon + 2000)) << 14 | day


with zipfile.ZipFile(ZIP) as archive, archive.open("occurrence.txt") as raw:
    text = io.TextIOWrapper(raw, encoding="utf-8", newline="")
    header = text.readline().rstrip("\r\n").split("\t")
    col = {name: i for i, name in enumerate(header)}
    need = ["gbifID", "acceptedTaxonKey", "decimalLatitude", "decimalLongitude",
            "coordinateUncertaintyInMeters", "eventDate", "license", "datasetKey", "recordedBy",
            "informationWithheld", "dataGeneralizations", "classKey", "class"]
    missing = [n for n in need if n not in col]
    if missing:
        raise SystemExit(f"columns missing: {missing}")
    idx = [col[n] for n in need]
    for line in text:
        if MAX_ROWS is not None and rows >= MAX_ROWS:
            break
        f = line.rstrip("\r\n").split("\t")
        rows += 1
        if rows % 250000 == 0:
            print(f"{rows} rows, {time.time() - started:.0f} s, RSS {rss_mb()} MB", file=sys.stderr,
                  flush=True)
        n = len(f)
        gid, taxon, lat, lon, unc, ev, lic, ds, by, withheld, gen, ck, cname = (
            f[i] if i < n else "" for i in idx)
        ck = ck.strip()
        if ck == "180":
            class_names_180[cname] += 1
        try:
            gid = int(gid)
            taxon = int(taxon)
        except ValueError:
            unloadable["gbifID or acceptedTaxonKey"] += 1
            continue
        latf, lonf = finite_float(lat), finite_float(lon)
        if latf is None or lonf is None or not (-90 <= latf <= 90 and -180 <= lonf <= 180):
            unloadable["coordinates"] += 1
            continue
        day = one_day(ev)
        if day is None:
            unloadable["eventDate not one calendar day"] += 1
            continue
        if day == "span":
            unloadable["eventDate spans more than one day"] += 1
            continue
        loaded += 1
        if withheld.strip() or gen.strip():
            dropped["not user-obscured"] += 1
            continue
        u = finite_float(unc)
        if u is None or u > 1000.0:
            dropped["uncertainty"] += 1
            continue
        lic = lic.strip()
        licences[lic] += 1
        by = by.strip()
        if not by:
            anonymous_by_dataset[ds.strip()] += 1
        obs = observer_id.setdefault(by, len(observer_id))
        d = dataset_id.setdefault(ds.strip(), len(dataset_id))
        try:
            la, lo = tenths(lat.strip()), tenths(lon.strip())
        except InvalidOperation:
            raise SystemExit(f"decimal text unreadable at gbifID {gid}: {lat!r} {lon!r}")
        doff = day.toordinal() - DAY_BASE
        assert 0 <= doff < (1 << 14)
        cc = lic in CC
        lichen = ck == "180"
        key = pack_key(taxon, obs, la, lo, doff)
        value = (gid << 20) | (d << 2) | (int(cc) << 1) | int(lichen)
        cur = kept.get(key)
        if cur is None or gid < (cur >> 20):
            kept[key] = value
        if cc:
            cur = kept_cc.get(key)
            if cur is None or gid < (cur >> 20):
                kept_cc[key] = value
        event_keys.add(((taxon << 11 | (la + 1000)) << 12 | (lo + 2000)) << 14 | doff)
        pre_outings.add(outing_key(obs, d, la, lo, doff))

after_filters = loaded - dropped["not user-obscured"] - dropped["uncertainty"]
print(f"stream done: {rows} rows, {time.time() - started:.0f} s, RSS {rss_mb()} MB",
      file=sys.stderr, flush=True)
summary = {
    "rows_read": rows,
    "loaded": loaded,
    "unloadable": dict(unloadable),
    "steps": {
        "not user-obscured": {"in": loaded, "dropped": dropped["not user-obscured"]},
        "uncertainty present and at most 1,000 m": {
            "in": loaded - dropped["not user-obscured"], "dropped": dropped["uncertainty"]},
        "after filters": after_filters,
        "observer key survivors": len(kept),
        "event key survivors (for comparison)": len(event_keys),
    },
    "licences_after_filters": dict(licences),
    "class_names_on_classKey_180_rows": dict(class_names_180),
    "distinct_observer_strings": len(observer_id) - 1,
    "datasets_after_filters": len(dataset_id),
}
del event_keys
# Could the two CC counts differ here at all? Keys where a CC record exists but the all-licence
# survivor is not CC; and anonymous keys held by more than one dataset.
summary["anonymous_records_after_filters_by_dataset"] = dict(anonymous_by_dataset)
summary["observer_keys_whose_survivor_is_not_cc_but_hold_a_cc_record"] = sum(
    1 for key in kept_cc if not (kept[key] >> 1) & 1)


def outings_from(survivors):
    """outing packed key -> flags: 1 any lichen survivor, 2 any CC survivor, 4 any CC lichen."""
    out = {}
    for key, value in survivors.items():
        obs, lat, lon, doff = unpack_key(key)
        ds = (value >> 2) & 0x3FFFF
        o = outing_key(obs, ds, lat, lon, doff)
        flags = value & 3
        if flags == 3:
            flags |= 4
        out[o] = out.get(o, 0) | flags
    return out


def in_whole_weeks(doff):
    return FIRST_MONDAY <= DAY_BASE + doff <= LAST_SUNDAY


summary["observer_key_survivors_in_whole_weeks"] = sum(
    1 for key in kept if in_whole_weeks(key & 0x3FFF))
out_all = outings_from(kept)
del kept
summary["cc_alone_observer_key_survivors"] = len(kept_cc)
out_cc_alone = outings_from(kept_cc)
del kept_cc
print(f"outings built, RSS {rss_mb()} MB", file=sys.stderr, flush=True)


def to_array(outings, cc_rule):
    """Columns: lat_tenths, lon_tenths, week index, day-type (0 weekend, 1 weekday), lichen,
    any-CC flag, day ordinal. Whole weeks only. cc_rule: keep only outings with a CC survivor,
    with lichen meaning a CC lichen survivor (the builder's rule)."""
    rows_ = []
    for o, flags in outings.items():
        if cc_rule and not (flags & 2):
            continue
        lichen = (flags >> 2) & 1 if cc_rule else flags & 1
        doff = o & 0x3FFF
        o >>= 14
        lon = (o & 0xFFF) - 2000
        o >>= 12
        lat = (o & 0x7FF) - 1000
        if not in_whole_weeks(doff):
            continue
        ordinal = DAY_BASE + doff
        week = (ordinal - FIRST_MONDAY) // 7
        weekday = (ordinal - 1) % 7  # 0 Monday (ordinal 1, 0001-01-01, is a Monday)
        rows_.append((lat, lon, week, 0 if weekday >= 5 else 1, lichen, (flags >> 1) & 1,
                      ordinal))
    return np.array(rows_, dtype=np.int32).reshape(-1, 7)


a_all = to_array(out_all, False)
a_cc_rule = to_array(out_all, True)
a_cc_alone = to_array(out_cc_alone, False)
summary["outings_before_duplicate_step_in_whole_weeks"] = sum(
    1 for o in pre_outings if in_whole_weeks(o & 0x3FFF))
summary["outings_after_duplicate_step_any_week"] = len(out_all)
lost = pre_outings.difference(out_all)
summary["outings_lost_to_the_observer_key"] = len(lost)
summary["outings_lost_to_the_observer_key_in_whole_weeks"] = sum(
    1 for o in lost if in_whole_weeks(o & 0x3FFF))
summary["outings_lost_named_vs_anonymous"] = {
    "named": sum(1 for o in lost if (o >> 37) >= (1 << 16)),
    "anonymous": sum(1 for o in lost if (o >> 37) < (1 << 16)),
}
del pre_outings, lost, out_all, out_cc_alone
np.save(OUT / "outings_all.npy", a_all)
np.save(OUT / "outings_cc_rule.npy", a_cc_rule)
np.save(OUT / "outings_cc_alone.npy", a_cc_alone)


def describe(a):
    cell_code = a[:, 0].astype(np.int64) * 10000 + a[:, 1]
    cells = np.unique(cell_code)
    cwd = np.unique((np.searchsorted(cells, cell_code) * 573 + a[:, 2]) * 2 + a[:, 3])
    cw = np.unique(cwd // 2)
    return {"outings": int(len(a)), "cells": int(len(cells)),
            "lichen_outings": int(a[:, 4].sum()),
            "nonzero_cell_week_daytype": int(len(cwd)), "cell_weeks_with_an_outing": int(len(cw)),
            "frame_rows_if_built": int(len(cells)) * 573 * 2}


def sorted_rows(a):
    b = a[:, [0, 1, 2, 3, 4, 6]]
    return b[np.lexsort(b.T[::-1])]


summary["all_licences"] = describe(a_all)
summary["cc_builder_rule"] = describe(a_cc_rule)
summary["cc_from_cc_records_alone"] = describe(a_cc_alone)
summary["cc_builder_rule_rows_equal_cc_alone_rows"] = bool(
    len(a_cc_rule) == len(a_cc_alone)
    and np.array_equal(sorted_rows(a_cc_rule), sorted_rows(a_cc_alone)))
summary["seconds"] = round(time.time() - started, 1)
summary["peak_rss_mb"] = resource.getrusage(resource.RUSAGE_SELF).ru_maxrss // 1024
(OUT / "recount_summary.json").write_text(json.dumps(summary, indent=2) + "\n")
print(json.dumps(summary, indent=2))
