"""Scratch: RSS over repeated t1_model.fit_predict (LightGBM) fits, no manual gc (synthetic only)."""
import os
import numpy as np
from forager_forecast import t1_model as tm

def cur():
    return round(int(open("/proc/self/statm").read().split()[1]) * os.sysconf("SC_PAGE_SIZE") / 2**20)

rng = np.random.default_rng(1)
x = rng.normal(size=(5000, 9)); y = (rng.random(5000) < 0.05).astype(float)
config = tm.tuning_configurations()[0]
print("config", {k: v for k, v in config.items()})
for k in range(60):
    tm.fit_predict(config, x, y, x, 2)
    if (k + 1) % 20 == 0:
        print("fit", k + 1, cur(), "MB", flush=True)
