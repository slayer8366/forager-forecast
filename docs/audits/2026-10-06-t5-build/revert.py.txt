def sha(p): return hashlib.sha256(Path(p).read_bytes()).hexdigest()

results = []
for name, path, old, new, tests in CHECKS:
    p = Path(path); saved = p.read_bytes(); saved_sha = sha(p)
    head = subprocess.run(["git", "show", f"HEAD:{path}"], capture_output=True).stdout
    text = saved.decode()
    assert text.count(old) == 1, f"{name}: edit anchor not unique or missing"
    xml = Path("/tmp/claude-1000/t5/revert.xml")
    xml.unlink(missing_ok=True)
    try:
        p.write_text(text.replace(old, new))
        for cache in Path(".").rglob("__pycache__"):
            if ".venv" not in cache.parts:
                shutil.rmtree(cache)
        env = {**os.environ, "PYTHONDONTWRITEBYTECODE": "1"}
        # The interpreter that runs the tests must see the edited source, or the run is stale.
        module = path.removeprefix("src/").removesuffix(".py").replace("/", ".")
        probe = subprocess.run(["uv", "run", "--frozen", "python", "-c",
                                f"import inspect, {module} as m; print({new!r} in inspect.getsource(m))"],
                               capture_output=True, text=True, env=env)
        edit_seen = probe.stdout.strip() == "True"
        run = subprocess.run(["uv", "run", "--frozen", "pytest", "-q", "-p", "no:warnings", "-p", "no:cacheprovider",
                              f"--junitxml={xml}", *tests], capture_output=True, text=True, env=env)
    finally:
        p.write_bytes(saved)
    restored_ok = sha(p) == saved_sha and p.read_bytes() == head
    out = run.stdout + run.stderr
    collection_error = (not edit_seen) or ("error during collection" in out) or ("ImportError" in out) or ("SyntaxError" in out) or not xml.exists()
    failures = []
    if xml.exists() and not collection_error:
        root = ET.parse(xml).getroot()
        for case in root.iter("testcase"):
            for f in list(case.findall("failure")) + list(case.findall("error")):
                msg = (f.get("message") or "").splitlines()[0][:160]
                failures.append(f"{case.get('name')}: {msg}")
    results.append({"check": name, "file": path, "tests": tests, "collection_error": collection_error,
                    "edit_seen_by_interpreter": edit_seen, "citable": not collection_error, "failed": len(failures), "failures": failures,
                    "restored_from_copy_and_equal_to_HEAD": restored_ok,
                    "tail": out.strip().splitlines()[-1] if out.strip() else ""})
    print(f"{name}: collection_error={collection_error} failed={len(failures)} restored={restored_ok}", flush=True)

Path(sys.argv[1]).write_text(json.dumps(results, indent=1) + "\n")
